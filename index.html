Auto-Execute Gold SR Break + Dot (2-4) on TradingView Paper Trading

TradingView Paper Trading executes trades automatically when you load a Pine Script v5 strategy() onto your chart (or connect its alert webhook to Termux). Copy the complete Pine Script strategy below, paste it into TradingView's Pine Editor, and click Add to Chart—it will automatically place 0.01 lot Buy/Sell orders with TP1, TP2, and SL on your TradingView Paper Trading account! 

//@version=5
strategy("AURA Gold VIP: SR Break + Dot (2-4) Auto-Trader", overlay=true, default_qty_type=strategy.fixed, default_qty_value=1, commission_type=strategy.commission.cash_per_order, commission_value=0.15)

// 1. ChartPrime SR Zone (20 Period Pivot Box) + Wavy Tunnel EMA 34
supZone = ta.lowest(low, 20)[1]
resZone = ta.highest(high, 20)[1]
ema20   = ta.ema(close, 20)
ema34   = ta.ema(close, 34)
rsi14   = ta.rsi(close, 14)
atr14   = ta.atr(14)

// 2. Exhaustion Dot Counter (Dots 1, 2, 3, 4)
var int bullDots = 0
var int bearDots = 0
isBullEx = (low <= supZone + atr14 * 0.50 and close > open) or (low < ema34 - atr14 * 0.70 and rsi14 < 45)
isBearEx = (high >= resZone - atr14 * 0.50 and close < open) or (high > ema34 + atr14 * 0.70 and rsi14 > 55)

if isBullEx
    bullDots += 1
    bearDots := 0
else if isBearEx
    bearDots += 1
    bullDots := 0
else
    if close > ema20
        bearDots := 0
    if close < ema20
        bullDots := 0

// 3. Exact User Rules: Dot #2-#4 Reversal OR Post-Dot #1 SR Break
buySignal  = (bullDots >= 2 and bullDots <= 4 and close >= open) or (bullDots[1] >= 1 and close > resZone and close > ema20)
sellSignal = (bearDots >= 2 and bearDots <= 4 and close <= open) or (bearDots[1] >= 1 and close < supZone and close < ema20)

if buySignal and strategy.position_size == 0
    strategy.entry("VIP_BUY", strategy.long, comment="Dot/SR Buy")
    strategy.exit("TP1", "VIP_BUY", qty_percent=50, limit=close + atr14 * 0.68, stop=close - atr14 * 1.35)
    strategy.exit("TP2", "VIP_BUY", limit=close + atr14 * 1.75, stop=close - atr14 * 1.35)

if sellSignal and strategy.position_size == 0
    strategy.entry("VIP_SELL", strategy.short, comment="Dot/SR Sell")
    strategy.exit("TP1", "VIP_SELL", qty_percent=50, limit=close - atr14 * 0.68, stop=close + atr14 * 1.35)
    strategy.exit("TP2", "VIP_SELL", limit=close - atr14 * 1.75, stop=close + atr14 * 1.35)
