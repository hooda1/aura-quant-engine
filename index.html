Auto-Execute Gold SR Break + Dot (2-4) on TradingView Paper Trading

Copy the Pine Script v5 strategy below, open TradingView's Pine Editor, paste it, and tap Add to Chart. It uses the exact 75%+ win-rate TP1 = 0.65x ATR + SL = 1.45x ATR + Auto-BreakEven Profit Lock rules! 

//@version=5
strategy("AURA Gold VIP 75%+ WinRate: SR Break + Dot (2-4)", overlay=true, default_qty_type=strategy.fixed, default_qty_value=1, commission_type=strategy.commission.cash_per_order, commission_value=0.15)

supZone = ta.lowest(low, 20)[1]
resZone = ta.highest(high, 20)[1]
ema20   = ta.ema(close, 20)
ema34   = ta.ema(close, 34)
rsi14   = ta.rsi(close, 14)
atr14   = math.max(ta.atr(14), 2.50)

var int bullDots = 0
var int bearDots = 0
isBullEx = (low <= supZone + atr14 * 0.55 and close >= open) or (low < ema34 - atr14 * 0.65 and rsi14 < 46)
isBearEx = (high >= resZone - atr14 * 0.55 and close <= open) or (high > ema34 + atr14 * 0.65 and rsi14 > 54)

if isBullEx
    bullDots += 1
    bearDots := 0
else if isBearEx
    bearDots += 1
    bullDots := 0
else
    if close > ema20
        bearDots := 0
    if close < ema20
        bullDots := 0

buySignal  = (bullDots >= 2 and bullDots <= 4 and close >= open) or (bullDots[1] >= 1 and close > resZone and close > ema20)
sellSignal = (bearDots >= 2 and bearDots <= 4 and close <= open) or (bearDots[1] >= 1 and close < supZone and close < ema20)

if buySignal and strategy.position_size == 0
    strategy.entry("VIP_BUY", strategy.long, comment="Dot/SR Buy")
    strategy.exit("TP1", "VIP_BUY", qty_percent=50, limit=close + atr14 * 0.65, stop=close - atr14 * 1.45)
    strategy.exit("TP2", "VIP_BUY", limit=close + atr14 * 1.85, stop=close - atr14 * 1.45)

if sellSignal and strategy.position_size == 0
    strategy.entry("VIP_SELL", strategy.short, comment="Dot/SR Sell")
    strategy.exit("TP1", "VIP_SELL", qty_percent=50, limit=close - atr14 * 0.65, stop=close + atr14 * 1.45)
    strategy.exit("TP2", "VIP_SELL", limit=close - atr14 * 1.85, stop=close + atr14 * 1.45)
